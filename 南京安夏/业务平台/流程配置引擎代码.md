/**
 * 统一API响应结果封装
 */
public class Result<T> {
    private int code;
    private String message;
    private T data;
    public Result setCode(ResultCode resultCode) {
        this.code = resultCode.code();
        return this;
    }

    public Result setCode(int code) {
        this.code = code;
        return this;
    }

    public int getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }

    public Result setMessage(String message) {
        this.message = message;
        return this;
    }

    public T getData() {
        return data;
    }

    public Result setData(T data) {
        this.data = data;
        return this;
    }

    @Override
    public String toString() {
        return JSON.toJSONString(this);
    }
}
/**
 * 响应码枚举，参考HTTP状态码的语义
 */
public enum ResultCode {
    SUCCESS(200),//成功
    FAIL(400),//失败
    UNAUTHORIZED(401),//未认证（签名错误）
    NOT_FOUND(404),//接口不存在
    INTERNAL_SERVER_ERROR(500);//服务器内部错误


    private final int code;

    ResultCode(int code) {
        this.code = code;
    }

    public int code() {
        return code;
    }
}

    /**
     * @Description:：单独保存某一步骤配置
     * @author：BoBo
     */
    @ApiOperation(value = "单独保存某一步骤配置")
    @PostMapping("/saveStepConfigSingle")
    public Result saveStepConfigSingle(@RequestParam String flowKey, @RequestBody JSONObject stepConfig) {
        Condition condition = new Condition(ConfActFlowinfo.class);
        Example.Criteria criteria = condition.createCriteria();
        criteria.andEqualTo("flowkey", flowKey);
        criteria.andEqualTo("isdeleted", 0);
        ConfActFlowinfo temp = new ConfActFlowinfo();
        List<ConfActFlowinfo> actFlowinfo = confActFlowinfoService.findByCondition(condition);
        if (actFlowinfo.size() == 0) {
            return ResultGenerator.genFailResult("该步骤暂未配置");
        }
        temp = actFlowinfo.get(0);
        JSONObject jsonObj = JSON.parseObject(temp.getFlowjson(), JSONObject.class);
        List<JSONObject> stepList = JSON.parseArray(jsonObj.get("steps").toString(), JSONObject.class);
        int index = 0;
        boolean flag = false;
        for (int i = 0; i < stepList.size(); i++) {
            // 找到当前要保存的步骤
            JSONObject step = stepList.get(i);
            if (null != step.get("id") && step.get("id").toString().equals(stepConfig.get("id").toString())) {
                index = i;
                flag = true;
                break;
            }
        }
        if (flag == true) {
            stepList.remove(index);
            stepList.add(stepConfig);
        }
        jsonObj.put("steps", stepList);
        temp.setFlowjson(jsonObj.toJSONString());
        confActFlowinfoService.update(temp);
        if (redis.hasKey("actflowinfo" + temp.getFlowkey(), 5)) {
            redis.del(5, "actflowinfo" + temp.getFlowkey());
            redis.set("actflowinfo" + temp.getFlowkey(), temp, 5);
        } else {
            redis.set("actflowinfo" + temp.getFlowkey(), temp, 5);
        }
        return ResultGenerator.genSuccessResult("保存成功");
    }

        for (JSONObject step : stepList) {
            if (null != step.get("id") && step.get("id").toString().equals(sid)) {
                if (null != isFilterStep && isFilterStep == true) {
                    stepConfig.put("fieldStatus", filterStep(step));

                    stepConfig.put("fieldStatus", step.getJSONArray("fieldStatus"));
                }
                step.remove("fieldStatus");
                stepConfig.put("stepConfig", step);
                stepConfig.put("flowSource", flowSource);
                stepConfig.put("personNumType", personNumType);
            }
            stepConfig.put("flowTitle", temp.getFlowtitle());
        }
        return ResultGenerator.genSuccessResult(stepConfig);
    }


    public List<JSONObject> filterStep(JSONObject step) {
        List<JSONObject> newArr = new ArrayList<>();
        JSONArray json = step.getJSONArray("fieldStatus");
        if (json.size() > 0) {
            for (int i = 0; i < json.size(); i++) {
                JSONObject obj = json.getJSONObject(i);  // 遍历 jsonarray 数组，把每一个对象转成 json 对象
                if (!obj.get("toggleStatus").toString().equals("0") || !obj.get("toggleCheck").toString().equals("0")) {
                    newArr.add(obj);
                }
            }
        }
        return newArr;
    }

    @ApiOperation(value = "treeActFlowinfo")
    @PostMapping("/tree")
    public Result trees(@RequestBody SearchConditionsModel searchConditionsModel, @RequestParam(required = false) String flag, @RequestParam(required = false) String type) {
        Auth auth = JwtUtils.GetAuths();
        Map<String, Object> params = new HashMap<>();
        CommRole role = roleService.findById(auth.getRoleID());
        String roleflowkeys = "";
        if (StringUtils.isNotEmpty(role.getFlowTemplateId())) {
            ConfActFlowinfoTemplate flowinfoTemplate = confActFlowinfoTemplateService.findById(role.getFlowTemplateId());
            roleflowkeys = flowinfoTemplate.getFlowKeys();
        } else {//防止未配置导致发起不了流程，使用老配置
            roleflowkeys = role.getRoleflowkeys();
        }
        List<FlowinfoDto> resultList = new ArrayList<>();
        if (!UserUtil.getAdminNames().contains(auth.getUserName()) && StringUtils.isEmpty(roleflowkeys)) {
            PageInfo pageInfo = new PageInfo(resultList);
            return ResultGenerator.genSuccessResult(pageInfo);
        }
        roleflowkeys = roleflowkeys == null?"":roleflowkeys;
        List<String> sids = Arrays.asList(roleflowkeys.split(","));
        SqlModel sqlModel = ConditionUtils.getConditionsPlus(searchConditionsModel);
        params.put("sqlModel", sqlModel);
        params.put("type", flag);
        params.put("sids",UserUtil.getAdminNames().contains(auth.getUserName()) ? null : sids);
        List<FlowinfoDto> flowinfoDtoList = new ArrayList<>();
        if("wenshuType".equals(type)){
            flowinfoDtoList = confActFlowinfoService.findFlowList(params);
        }else {
            flowinfoDtoList = confActFlowinfoService.findFlowListByType(params);
        }
        if(flowinfoDtoList.size() == 0){
            PageInfo pageInfo = new PageInfo(resultList);
            return ResultGenerator.genSuccessResult(pageInfo);
        }
        List<String> codeTypeList = flowinfoDtoList.stream().sorted(Comparator.comparing(FlowinfoDto::getCodetypeorder)).map(f -> f.getCodetype()).distinct().collect(Collectors.toList());
        for(String codeType : codeTypeList){
            List<String> codeValueList = flowinfoDtoList.stream().sorted(Comparator.comparing(FlowinfoDto::getCodeorder)).filter(f -> f.getCodetype().equals(codeType)).map(f -> f.getCodevalue()).distinct().collect(Collectors.toList());
            List<FlowinfoDto> list = new ArrayList<>();
            if(codeValueList.size() > 0){
                for(String codeValue : codeValueList){
                    List<FlowinfoDto> flowinfoDtos = flowinfoDtoList.stream().filter(f -> f.getCodevalue().equals(codeValue)).collect(Collectors.toList());
                    if(flowinfoDtos.size() > 0){
                        list.add(new FlowinfoDto(codeValue, null, flowinfoDtos,""));
                    }
                }
                resultList.add(new FlowinfoDto(codeType, null, list,""));
            }
        }
        PageInfo pageInfo = new PageInfo(resultList);
        return ResultGenerator.genSuccessResult(pageInfo);
    }

                        return ResultGenerator.genFailResult("退回失败");
                    }
                }
                confActFlowinfoService.deleteSubFlowInfo(myTask.getBusinessID(),myTask.getFlowKey());
                platformTransactionManager.commit(status);
                return ResultGenerator.genSuccessResult("删除成功");
            }
            String processinstanceid = myTask.getProcessInstanceId();
            //获取下一步节点
            FlowElement nextStep = flowableUtil.getFlowElementByStepID(myTask.getProcessDefinitionId(), newStepId);

            //获取流程的下一骤的配置
            FlowJson.Step nextstepCnf = confActFlowinfoService.getStepConf(myTask.getFlowKey(), nextStep.getId());
//            FlowJson.Step curstepCnf = confActFlowinfoService.getStepConf(myTask.getFlowKey(), myTask.getFlowElementId());
            List<ActTaskApproveLoopsetting> loopsettings = null;
            UserTask userTask = (UserTask) nextStep;

    private String judgeBusinessId(String businessID) {
        String clueid = "";
        BusActClueinfo busActClueinfo = busActClueinfoService.findById(businessID);
        if(null != busActClueinfo){
            if(null == busActClueinfo.getSourceType()){
                return clueid;
            }else{
                if(5==busActClueinfo.getSourceType()){
                    //径送上级标记（下发上逻辑），措施流程下看上,BusReClueClue的ClueId为businessID
                    BusReClueClue busReClueClue = busReClueClueService.findBy("clueId",businessID);
                    if(null != busReClueClue){
                        clueid =  busReClueClue.getClueIdRe();
                    }
                }
                if(4==busActClueinfo.getSourceType()){
                    //交办下级标记（上发下逻辑），主流程上看下,BusReClueClue的ClueIdRe 为businessID
                    BusReClueClue busReClueClue = busReClueClueService.findBy("clueIdRe",businessID);
                    if(null != busReClueClue){
                        clueid =  busReClueClue.getClueId();
                    }
                }
            }
        }
        return  clueid;
    }

    @PostMapping("/transferStep")
    @ApiOperation(value = "流程跳转/驳回/退回")
    @MyLog(value = "流程跳转/驳回/退回")
    @PushData
    @ClueRabbitMQAnnotation
    @UpdateBaseUndertaker
    public Result transferStep(@RequestBody MyTask myTask) {
        //开启事务管理
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        TransactionStatus status = platformTransactionManager.getTransaction(def);
        try {
            Auth auth = JwtUtils.GetAuths();
            String processinstanceid = myTask.getProcessInstanceId();
            CommUsers commUsers = commUsersService.findById(myTask.getApproveuser());

            if(commUsers==null){
                return ResultGenerator.genFailResult("转发失败,转发人不存在！");
            }

            //添加任务跳转的delete_reason描述
            managementService.executeCommand(new TaskRecordCmd(myTask.getTaskid(), "transfer"));

            boolean isHandleStep = false;

            if(myTask.getIsapprove().equals("1"))
            {
                int curLoop = (int) runtimeService.getVariable(myTask.getExecutionId(),"loopCounter");
                managementService.executeCommand(new ApproveBackCmd(myTask.getExecutionId(), curLoop-1));
                Map<String,Object> map = new HashMap<>();
                map.put(flowReceiverId, myTask.getApproveuser());
                taskService.complete(myTask.getId(), map);
                //更换审批步骤配置人信息
                actTaskApproveLoopsettingService.changeApproveUserId(myTask.getApproveuser(), myTask.getExecutionId(),auth.getUserID());
            }else{
                //单例模式，配置候选人，candidateUser
                runtimeService.setVariable(myTask.getExecutionId(), flowSendSingleInstnce, myTask.getApproveuser());
                if (myTask.getSendType() == 7) {
                    // 转B角状态修改，防止监听器搜索到send重复操作
                    runtimeService.setVariable(myTask.getExecutionId(), flowSendType, "AToB");
                }
                //步骤跳转
                runtimeService.createChangeActivityStateBuilder()
                        .processInstanceId(processinstanceid)
                        .moveActivityIdTo(myTask.getFlowElementId(), myTask.getFlowElementId())
                        .changeState();
                //更换承办人id
                confActFlowinfoService.updateHiTaskAssignee(auth.getUserID(), myTask.getId());
            }



            //查询最新runtask
            Task singleResult = taskService.createTaskQuery().processInstanceId(myTask.getProcessInstanceId()).singleResult();
            if(singleResult != null)
            {
                //获取新步骤信息返回给前端
                MyTask newMytask = confActFlowinfoService.backMyTask(singleResult.getId());
                if(newMytask != null){
                    //更换承办人id
                    confActFlowinfoService.updateHiTaskAssignee(commUsers.getId(), newMytask.getId());
                    //添加task记录
                    busActTaskService.addTask(singleResult.getId(), myTask.getBusinessID(), myTask.getApproveLoopSettingId(), auth, commUsers, myTask.getFlowKey(), myTask.getId(), myTask.getId() ,myTask.getBusFormId());

                    busActTaskService.updateTask(myTask.getId(), newMytask.getId(), myTask.getApproveuser(), myTask.getBusinessID());
                    //更新businessid映射表中的当前bustaskid和busformid
                    busReBusFlowkeyService.updateCurMsg(newMytask.getId(), "", newMytask.getBusinessID(), singleResult.getProcessInstanceId(), myTask.getFlowKey());
                    //添加转发人与业务关系
                    busReUserBusService.addRe(newMytask.getBusinessID(), commUsers.getId(), commUsers.getDeptid(), commUsers.getCompanyid(),FlowableConstant.UserBusType.deaf);
                    platformTransactionManager.commit(status);
                    WebSocketServer.sendInfo("refreshMenu", auth.getUserName());
                    WebSocketServer.sendInfo("refreshMenu", commUsers.getUsername());
                    return ResultGenerator.genSuccessResult(newMytask);
                }
            }
            return ResultGenerator.genFailResult("转发失败！");


        } catch (Exception e) {
            platformTransactionManager.rollback(status);
            e.printStackTrace();
            return ResultGenerator.genFailResult("转发失败！");
        }

    /**
     * @Description:：返回所有步骤信息
     * @param：flowKey
     * @author：CG
     */
    @PostMapping("/stepInfo")
    @ApiOperation(value = "返回所有步骤信息（配置trend）")
    public Result stepInfo(@RequestParam String flowKey, @RequestParam String flowElementId) {
        List<NextStep> nextSteps = new ArrayList<>();
        List<NodeMessage> nodeMessages = new ArrayList<>();
        //id为流程定义的key名（如clueinfo01）
        FlowProcdefDto proDef = confActFlowinfoService.findProDefId(flowKey);
        //获取流程Model
        BpmnModel bpmnModel = repositoryService.getBpmnModel(proDef.getId());
        Process process = bpmnModel.getProcesses().get(0);
        Collection<FlowElement> flowElements = process.getFlowElements();
        for (FlowElement fe : flowElements) {
            //获取用户节点
            if (fe instanceof UserTask) {
                UserTask userTask = (UserTask) fe;
                NodeMessage nodeMessage = new NodeMessage();
                nodeMessage.setNodeId(userTask.getId());
                if (null != userTask.getDocumentation()) {
                    nodeMessage.setNodename(userTask.getDocumentation());
                } else {
                    nodeMessage.setNodename(userTask.getName());
                }
                nodeMessage.setNodekey(userTask.getDocumentation());
                nodeMessages.add(nodeMessage);
            }
            //获取线节点
            if (fe instanceof SequenceFlow) {
                SequenceFlow s = (SequenceFlow) fe;
                NextStep nextStep = new NextStep();
                if (flowElementId.equals(s.getSourceRef())) {
                    nextStep.setTargetRef(s.getTargetRef());
                    nextStep.setTargetName(s.getTargetFlowElement().getName());
                    nextStep.setWireId(s.getId());
                    nextStep.setSourceName(s.getSourceFlowElement().getName());
                    nextStep.setSourceRef(s.getSourceRef());
                    nextStep.setWireName(s.getName());
                    nextSteps.add(nextStep);
                }
            }
        }
        return ResultGenerator.genSuccessResult(nextSteps);
    }

    @ApiOperation(value = "发布任务/完成任务")
    @PostMapping(value = "/sendFlowMyTask")
    @MyLog(value = "发布任务/完成任务")
    @AddNoticeBusMsg(type = 3)
    @PushData
    @ClueRabbitMQAnnotation
    public Result sendFlowMyTask(@RequestBody MyTask task) {
        //获取当前任务id
        String taskId = task.getId();

        String oldFormNmae = "";

        //任务id为空情况
        if (StringUtils.isEmpty(task.getId())) {
            return ResultGenerator.genFailResult("任务Id为空");
        }
        //未选择发送人情况
        if (null == task.getApproveuser()) {
            return ResultGenerator.genFailResult("请先设置分配人!");
        }
        //根据当前任务id查找最新runtask任务相关信息
        Task newtask = taskService.createTaskQuery().taskId(taskId).singleResult();
        //如果不存在说明当前taskid对应任务已经发走
        if (null == newtask) {
            return ResultGenerator.genFailCodeResult(700, "当前事项已被处理");
        }

        //开启事务管理
        DefaultTransactionDefinition def = new DefaultTransactionDefinition();
        def.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRED);
        TransactionStatus status = platformTransactionManager.getTransaction(def);

        boolean isMultiInstance = false;

        try {
            //从token获取当前用户信息
            Auth auth = JwtUtils.GetAuths();

            CommUsers commUsers = commUsersService.findById(task.getApproveuser());
            CommGrantUsers commGrantUsers = commGrantUsersService.findByOriginatorId(commUsers.getId());

            //判断是否有授权人
            boolean isGrant = false;
            if (commGrantUsers != null && StringUtils.isNotBlank(commUsers.getCandidateuser())){
                isGrant = true;
                commUsers = commUsersService.findById(commGrantUsers.getAgentId());
                task.setApproveuser(commGrantUsers.getAgentId());
            }

            Auth authApproveuser = new Auth(commUsers);

            //工作流发送参数
            HashMap<String, Object> map = new HashMap<>();

            //如果当前任务没有办理人，则该步骤配置为候选人，需赋值办理人
            if (StringUtils.isEmpty(newtask.getAssignee())) {
                taskService.setAssignee(taskId, auth.getUserID());
            }
            //初始化processInstanceid
            String processInstanceId = task.getProcessInstanceId();
            //流程key
            String flowKey = task.getFlowKey();
            //当前步骤是否为子流程步骤
            boolean isSubProcessStep = false;
            //当前审批配置是否已经生成
            boolean isLoopSetting = false;
            //定义businessid
            String businessid = task.getBusinessID();
            //定义审批步骤节点
            List<ActTaskApproveLoopsetting> loops = null;
            //审批步骤loopsettingid
            String loopSettingid = StringUtils.isNotEmpty(task.getApprovestepid()) ? task.getApprovestepid() : "";
            //定义下一步的流程json
            FlowJson.Step nextstepCnf = null;
            //获取流程的当前步骤的配置
            FlowJson.Step curStepCnf = confActFlowinfoService.getStepConf(flowKey, task.getFlowElementId());
            oldFormNmae = curStepCnf.getFormTitle();
            //获取当前的BusReBusFlowkey信息
            BusReBusFlowkey fromBrf = busReBusFlowkeyService.findById(businessid);
            //定义busForm
            BusForm busForm = new BusForm();
            //设置主流程businessId临时变量
            map.put("mainflowBusinessId", task.getBusinessID());
            map.put("mainflowKeysId", flowKey);
            map.put("jumpStepIndex",task.getJumpStepIndex());
            map.put("handleTypeDate", task.getHandleTypeDate());
            FlowElement nextStep = null;
            CallActivity callActivity = null;
            //如果当前步骤非审批步骤，或当前步骤为审批通过，则获取下一步骤的信息
            if ((!"1".equals(task.getIsapprove())) || 3 == task.getSendType() || 4 == task.getSendType()) {
                //非审批通过不获取下一步骤节点
                //todo 领导审批通过自动下一步骤 需要不能判断下一步审批
                if(3 != task.getSendType()){
                    nextStep = flowableUtil.nextSequenceFlow(task.getProcessDefinitionId(), task.getFlowElementId(), task.getTrend());
                    if( nextStep instanceof CallActivity)
                    {
                        callActivity = (CallActivity) nextStep;
                        flowKey = callActivity.getCalledElement();
                        nextStep = flowableUtil.getFirstElement(callActivity.getCalledElement(), task.getTrend());
                        isSubProcessStep = true;
                    }
                    nextstepCnf = confActFlowinfoService.getStepConf(flowKey, nextStep.getId());
                    if(null != nextstepCnf && nextstepCnf.getProcessDate()!=0){
                        //添加到期时间
                        map.put("dueDate", "P" + nextstepCnf.getProcessDate() + "D");
                    }
                }

                //如果下一步为null，并且当且流程为子流程
                if (null == nextStep && 3 != task.getSendType() && StringUtils.isNotEmpty(task.getSuperProcessInstanceId())) {
                    Map<String, String> runActMap = confActFlowinfoService.getProIdActIdByCallProInstId(task.getProcessInstanceId());
                    nextStep = flowableUtil.nextSequenceFlow(runActMap.get("procDefId"), runActMap.get("actId"), task.getTrend());
                    flowKey = confActFlowinfoService.selectFlowkeyByFlowId(runActMap.get("procDefId"));
                    processInstanceId = task.getSuperProcessInstanceId();
                }
                //若当前操作为发送至下一步骤，且下一步是子流程，则flowkey获取子流程的key
                else if (3 != task.getSendType() && isSubProcessStep) {
                    //若不继承businesskey，则添加新变量
                    if (!callActivity.isInheritBusinessKey()) {
                        //设置子流程的businessId临时变量
                        businessid = UUID.randomUUID().toString();
                        map.put("subflowBusinessId", businessid);
                        //添加businessid和flowkey关联关系
                        map.put("auditFileNameList",task.getAuditFileNameList());
                        map.put("affixid",task.getAffixid());
                        busReBusFlowkeyService.addBusFlowkey(businessid, flowKey,fromBrf.getIsZaz(),fromBrf.getZazDeptId(),authApproveuser);
                    }
                    //设置子流程的flowkey临时变量
                    map.put("subflowKey", flowKey);
                }
                isMultiInstance = flowableUtil.isMultiInstance(nextStep);



                //如果当前步骤非审批步骤 或者 下一步如果配置了多例模式且为审批通过
                if (3 != task.getSendType() && isMultiInstance) {
                    //若不为子流程步骤
                    isLoopSetting = true;
                    //若还未配置审批，则获取跳转步骤配置的审批人员信息
                    loops = actTaskApproveLoopsettingService.setActTaskApproveLoopsettings(businessid, flowKey, nextStep.getId(), task.getApproveuser());
                    //获取approvesetting的id当做接下来处理步骤id
                    List<String> stepids = loops.stream().map(l -> l.getId()).collect(Collectors.toList());
                    //赋值下一步审批的审批配置id
                    loopSettingid = stepids.get(0);
                    //设置多例审批id
                    map.put(flowSendMultiInstnce, stepids);
                } else {
                    //单例模式，配置候选人，candidate
                    map.put(flowSendSingleInstnce, task.getApproveuser());
                }

                //如果当前发送为审批通过
                if (task.getSendType() == 3 || task.getSendType() == 4) {
                    //下一步不为多例模式则不需要设置loopsetting
                    if (!isMultiInstance) {
                        loopSettingid = "";
                    }
                    //审批通过分为两种：
                    // 1.配置为"审批通过自动下一步"，则无论承办人还是领导审批，都全部到下一步
                    // 2.配置为"非审批通过自动下一步"，需要交还承办人，则再分为两种情况：a.承办人步骤审批通过。则相当于纸质呈批，直接到下一步；b.领导在审批过程中审批通过，则为发还承办人，回到串行审批的第一个步骤；
                    //若当前步骤为非审批通过自动下一步且不是审批阶段最后一步，或者当前步骤为承办人办理步骤，则都将当前节点跳转至当前审批环节最后一步并发送
                    List<ActTaskApproveLoopsetting> curloops = actTaskApproveLoopsettingService.getApproveLoopListByExecutionId(task.getExecutionId());
                    if (curStepCnf.getAutoSendNextStep() || task.getIsApproveHandleStep()) {
                        managementService.executeCommand(new ApproveBackCmd(task.getExecutionId(), curloops.size()));
                    } else //跳转至审批阶段第一步办理步骤
                    {
                        loopSettingid = curloops.get(0).getId();
                        managementService.executeCommand(new ApproveBackCmd(task.getExecutionId(), -1));
                    }
                    //标识当前步骤已经通过审批
                    actTaskApproveLoopsettingService.updateApproveStatus(task.getFlowElementId(), task.getProcessInstanceId());
                    //通过审批后生成编号
                    if ("立案呈批报告".equals(task.getNodename()) || "初步核实方案审批".equals(task.getNodename())) {
                        //生成立案以及初核编号
                        busActCluecodeService.addCLCode(task);
                    }
                    //若为领导审批通过，则记录该次审批通过时间
                    if (task.getSendType() == 3) {
                        busFormService.updateApproveMsg(task.getApproveLoopSettingId(),auth.getUserID(), auth.getRealName(), curStepCnf.getFormDesignId(), businessid);
                    }

                }


            } else {
                if ("1".equals(task.getIsapprove())) {
                    managementService.executeCommand(new ApproveBackCmd(task.getExecutionId(), task.getJumpStepIndex() - 1));
                    //判断承办人是否需要更换
                    if(StringUtils.isNotBlank(task.getApprovestepid()))
                    {
                        actTaskApproveLoopsettingService.judgeAndUpdateApproveUserIds(task.getApproveuser(),task.getApprovestepid());
                    }
                }

                //单例模式，配置候选人，candidate
                map.put(flowSendSingleInstnce, task.getApproveuser());
            }

            //设置当前步骤businessId临时变量
            map.put("curBusinessId", task.getBusinessID());

            //前端给的发送人
            map.put(flowReceiverId, task.getApproveuser());

            //设置临时变量 发送类型为发送
            if (task.getIsapprove().equals("1") && task.getSendType() != 4 && !curStepCnf.getAutoSendNextStep()) {
                map.put(flowSendType, "approveSend");
            }
            else {
                map.put(flowSendType, "send");
            }

            //配置trend
            if (StringUtils.isNotEmpty(task.getTrend()))
                map.put(flowTrend, task.getTrend());
                //如果当前发送为审批退回，则添加退会标识deletereason
            else if (task.getSendType() == 2) {
                //设置临时变量 发送类型为退回
                map.put(flowSendType, "approveBack");
            }
            if (task.getSendType() == 7) {
                map.put(flowSendType, "AToB"); // 转B角状态修改，防止监听器搜索到send重复操作
            }
            if (task.getSendType() == 2) {
                managementService.executeCommand(new TaskRecordCmd(task.getTaskid(), flowBackReason));
            }

            //当前步骤非审批发送，下一步骤有新表单，则添加新表单信息,添加duedate
            if (task.getSendType() != 3 && null != nextStep && !(nextStep instanceof EndEvent)) {
                busForm = busFormService.addNewForm(businessid, task.getApproveuser(), flowKey, nextstepCnf, task.getNeedNewSpb()==null?false:task.getNeedNewSpb());
            }
            else{
                //如果当前不需要生成新表单，则获取当前的表单id信息
                busForm = busFormService.findById(task.getBusFormId());
            }
            //添加当前承办部门临时变量
            map.put("curSubmitDept", busForm.getSubmitDeptId());
            //添加当前时间
            map.put("curTime", DateUtil.date());
            //添加下一步loopsetid变量
            map.put("nextLoopSetId", loopSettingid);

            //完成当前步骤
            taskService.complete(taskId, map);

            //如果下一步为子流程，则需要获取子流程的processInstanceid
            if (isSubProcessStep) {
                ProcessInstance processInstance = runtimeService.createProcessInstanceQuery().superProcessInstanceId(task.getProcessInstanceId()).singleResult();
                if (null != processInstance) {
                    processInstanceId = processInstance.getProcessInstanceId();
                    //更新子流程状态(标题，流程名，key)
                    List<ConfActFlowinfo> actflowinfo = confActFlowinfoService.selectByFlowkey(flowKey);
                    ConfActFlowinfo temp = actflowinfo.get(0);
                    actGeBytearrayService.updateHiProcinstFlow(StringUtils.isNotEmpty(temp.getLabel()) ? temp.getLabel() : temp.getFlowname(), processInstance.getProcessDefinitionKey(), processInstance.getProcessInstanceId());
                    /*String flowtitle = task.getFlowtitle();
                    if (StringUtils.isNotEmpty(flowtitle) && flowtitle.contains("【")) {
                        String substring = flowtitle.substring(flowtitle.indexOf("【") + 1, flowtitle.indexOf("】"));
                        String newFlowTitle = flowtitle.replaceAll(substring, temp.getLabel());
                        busReBusFlowkeyService.updateFlowTitle(processInstance.getBusinessKey(), newFlowTitle);
                    }*/
                }
            }
            //是否子流程發還主流程
            Boolean isSubFlowSendBack = false;
            //查找发送后的最新步骤
            Task singleResult = taskService.createTaskQuery().processInstanceId(processInstanceId).singleResult();
            //若找不到下一步，有三种情况
            //1.普通流程办结归挡
            //2.子流程办结归档，下一步不发回主流程
            //2.子流程办结归档，下一步为发回主流程
            //子流程需要获取父级processinstenceid，查询当前是否有新待办事项，决定是否需要发回主流程
            if(singleResult == null){
                HistoricProcessInstance historicProcessInstance = historyService.createHistoricProcessInstanceQuery().processInstanceId(processInstanceId).singleResult();
                if(historicProcessInstance!=null && historicProcessInstance.getSuperProcessInstanceId()!=null){
                    singleResult = taskService
                            .createTaskQuery()
                            .processInstanceId(historicProcessInstance.getSuperProcessInstanceId())
                            .singleResult();
                    isSubFlowSendBack = true;
                }
            }


            //如果有审批配置，则添加审批配置(同时判断是否为退回再审批情况，避免重复插入数据)

                actTaskApproveLoopsettingService.addSettings(loops, processInstanceId,singleResult.getExecutionId());

    @PostMapping("/getCommentBox")
    public Result GetCommentBox(@RequestBody MyTask task, @RequestParam(required = false, defaultValue = "") String editFlag) {
        Auth auth = JwtUtils.GetAuths();
        //初始化意见栏集合
        List<CommentBox> cmList = new ArrayList<>();
        //获取该步骤配置
        FlowJson.Step stepCnf = confActFlowinfoService.getStepConf(task.getFlowKey(), task.getFlowElementId());
        //若该意见栏配置为获取某个步骤意见栏，则赋值配置的步骤stepid
        List<String> stepIdList = new ArrayList();
        //查询意见的流程实例id
        String procInstId = task.getProcessInstanceId();
        //获取配置的审批步骤
        List<ActTaskApproveLoopsetting> allLoops = new ArrayList<>();
        //是否看其他步骤意见
        Boolean hasOtherStepOpninon = false;
        //是否有自定义审批意见栏顺序
        Boolean hasApproveOrder = stepCnf.getHasApproveOrder() == null?false:stepCnf.getHasApproveOrder();
        //判断步骤意见栏配置：展示父级流程步骤 / 展示某步步骤
        if(stepCnf.getIsShowParentOpinion() != null && stepCnf.getIsShowParentOpinion())
        {
            hasOtherStepOpninon = true;
            //查找父级流程最后一步的stepid，根据步骤配置，查找该展示的意见栏
            String superProcInstId = actHiProcinstService.getSuperProcessInstanceIdById(task.getProcessInstanceId());
            if(StringUtils.isNotBlank(superProcInstId))
            {
                procInstId = superProcInstId;
                ActHiTaskinst lastTask = actHiTaskinstService.getLastTaskBypRrocInstId(superProcInstId);
                ActHiProcinst actHiProcinst = actHiProcinstService.findById(superProcInstId);
                if(lastTask!=null)
                {
                    FlowJson.Step pStepCnf = confActFlowinfoService.getStepConf(actHiProcinst.getFlowkey(), lastTask.getTaskDefKey());
                    hasApproveOrder = pStepCnf.getHasApproveOrder() == null?false:pStepCnf.getHasApproveOrder();
                    if(pStepCnf.getOpinionStepId()!=null){
                        String[] opinionStepIdArr = pStepCnf.getOpinionStepId().replaceAll("\\[","").replaceAll("\\]","").split(",");
                        stepIdList = new ArrayList<>(Arrays.asList(opinionStepIdArr));
                    }
                    if(stepIdList.size()!=0)
                    {
                        allLoops = actTaskApproveLoopsettingService.getApproveLoopListByStepIds(stepIdList,actHiProcinst.getBusinessKey());
                    }
                }
            }
        }
        else
        {
            List<String> opinionSteps = new ArrayList<>();
            if(stepCnf.getOpinionStepId()!=null){
                hasOtherStepOpninon = true;
                String[] opinionStepIdArr = stepCnf.getOpinionStepId().replaceAll("\\[","").replaceAll("\\]","").split(",");
                opinionSteps = new ArrayList<>(Arrays.asList(opinionStepIdArr));
                stepIdList.addAll(opinionSteps);
            }
            if(!stepIdList.contains(task.getFlowElementId()))
                stepIdList.add(task.getFlowElementId());
            allLoops = actTaskApproveLoopsettingService.getApproveLoopListByStepIds(stepIdList,task.getBusinessID());

            //如果只查看该步骤对应的某些审批环节
            if(opinionSteps.size() == 1 && StringUtils.isNotBlank(stepCnf.getOpinionLoopSortId()))
            {
                List<Integer> orders = new ArrayList<>();
                String[] loopOrder = stepCnf.getOpinionLoopSortId().split(",");
                //由于loops有承办人步骤，因此每个步骤加1
                for(String sort:loopOrder)
                {
                    orders.add(Integer.parseInt(sort)+1);
                }
                String oStepId = opinionSteps.get(0);
                //筛选非所看步骤的所有loops及所看步骤匹配sort的loops
                allLoops = allLoops.stream().filter(a-> (a.getFlowelementid().equals(oStepId) && orders.contains(a.getSort()) || !a.getFlowelementid().equals(oStepId))).collect(Collectors.toList());
            }
        }

        List<ActTaskApproveLoopsetting> loops = new ArrayList<>();
        //筛选最新阶段的loops
        for(String id :stepIdList)
        {
//            FlowJson.Step curStepCnf = confActFlowinfoService.getStepConf(task.getFlowKey(), id);
            //取当前步骤最后一个审批节点
            ActTaskApproveLoopsetting curLoop = allLoops.stream().filter(l->l.getFlowelementid().equals(id)).sorted(Comparator.comparing(ActTaskApproveLoopsetting::getTimestamp).reversed()).findFirst().orElse( null );
            if(curLoop != null)
            {
                List<ActTaskApproveLoopsetting> curLoops = allLoops.stream().filter(l->l.getExecutionid().equals(curLoop.getExecutionid())).collect(Collectors.toList());
                loops.addAll(curLoops);
            }
        }

        Map<String,Map<String,Object>> audioMap = busAudioService.findAudioCount(task.getFlowElementId(),task.getProcessInstanceId());

        if ((!judgeApprove) && loops.size() > 0) {
            //获取当前步骤loop
            ActTaskApproveLoopsetting culoop = null;
            if (StringUtils.isNotBlank(task.getApprovestepid())) {
//                culoop = loops.stream().filter(l -> l.getId().equals(task.getApprovestepid())).collect(Collectors.toList()).get(0);
                loops.stream().filter(l -> l.getId().equals(task.getApprovestepid())).collect(Collectors.toList());
                if (loops.size()>0){
                    culoop = loops.get(0);
                }
            }
            //获取根据职级权限判断后的剩余可发送人
            Map<String, Object> approveMap = confActFlowinfoService.judgeLowestApprove(task, loops, stepCnf, culoop, 2);
            //获取审批权限后的意见栏
            if (null != approveMap.get("loops")) {
                loops = (List<ActTaskApproveLoopsetting>) approveMap.get("loops");
            }
        }
        //共享意见栏label数量
        int labelSize = 1;
        int lastLabelIndex = 0;
        //去除非审批步骤
        loops = loops.stream().filter(l -> StringUtils.isNotEmpty(l.getLabelnode()) && l.getNeedComment()).collect(Collectors.toList());
        if (loops.size() > 0) {
            //将审批步骤转为意见栏
            cmList = loops.stream().map(loop -> new CommentBox(loop)).collect(Collectors.toList());
            //将当前步骤设置为可编辑
            if ("1".equals(editFlag)) {
                cmList.stream().forEach(s -> s.setEdit(s.getApproveloopId().equals(task.getApprovestepid())));
            }
            //获取历史审批意见
            List<ActHiComment> comments = actHiCommentService.getCommentLabelsByIds(procInstId);
            //遍历意见栏，回填历史意见
            for (int i=0;i< cmList.size();i++) {
                CommentBox cm = cmList.get(i);
                //获取该意见栏的历史意见
                List<ActHiComment> currentComments = comments.stream().filter(c ->  c.getFlowElementId().equals(cm.getStepid()) && c.getApproveDeptId().equals(cm.getApproveDeptid()) && ( cm.getIsHandleStep() && c.getIsHandleStep() || c.getBookmark().equals(cm.getBookmark()))).collect(Collectors.toList());
                //回填语音数量
                if(audioMap.containsKey(cm.getApproveloopId())){
                    cm.setAudioCount(new Integer( audioMap.get(cm.getApproveloopId()).get("stepCount").toString() ));
                }
                if(i>=1 && cm.getLabel().equals(cmList.get(i-1).getLabel()))
                {
                    cm.setHasLabel(false);
                    labelSize ++;
                }else{
                    cm.setHasLabel(true);
                    cmList.get(lastLabelIndex).setLabelSize(labelSize);
                    labelSize = 1;
                    lastLabelIndex = i;
                }
                if(i == cmList.size()-1)
                {
                    cmList.get(lastLabelIndex).setLabelSize(labelSize);
                }
                if(cm.getApproveloopId().equals(task.getApprovestepid()))
                {
                    //如果是代录，则当前用户为书记
                    if(cm.getIsAssigneeSign())
                    {
                        //获取当前接收用户companyid
                        CommUsers mainLeader = commUsersService.findMainLeaderByUserId(cm.getApproveuserid());
                        if(mainLeader!=null)
                        {
                            cm.setRealname(mainLeader.getRealname());
                        }else{
                            cm.setRealname(mainLeader.getRealname());
                        }
                    }
                    else{
                        cm.setRealname(auth.getRealName());
                    }
                }
                //该意见栏无历史意见则跳过
                if (currentComments.size() == 0)
                    continue;
                //初始化历史意见
                List<CommentBox> children = new ArrayList<>();
                //如果该意见栏为编辑状态，则将该意见栏的所有意见放入历史意见
                if (cm.getEdit() && !cm.getIsHandleStep() && !task.getId().equals(currentComments.get(0).getTaskId())) {
                    children = currentComments.stream().map(c -> new CommentBox(c)).collect(Collectors.toList());
                } else {
                    //非编辑状态，则将最后一次意见赋值到当前意见
                    if (currentComments.get(0).getFullMsg() != null) {
                        cm.setComment(new String(currentComments.get(0).getFullMsg()));
                    }
                    cm.setTime(currentComments.get(0).getTime());

                    //如果是代录，则当前用户为书记
                    if(cm.getIsAssigneeSign())
                    {
                        //获取当前接收用户companyid
                        CommUsers mainLeader = commUsersService.findMainLeaderByUserId(cm.getApproveuserid());
                        if(mainLeader!=null)
                        {
                            cm.setRealname(mainLeader.getRealname());
                        }else{
                            cm.setRealname(currentComments.get(0).getUsername());
                        }
                    }
                    else{
                        cm.setRealname(currentComments.get(0).getUsername());
                    }

                    //若该意见栏还有其他意见，则将其他意见放入该步骤的历史意见
                    if (currentComments.size() > 1) {
                        //跳过index=0的最新步骤的意见
                        for (int j = 1; j < currentComments.size(); j++) {
                            children.add(new CommentBox(currentComments.get(j)));
                        }
                    }
                }
                //赋值历史意见
                cm.setHistoryComment(children);

            }
            //意见栏排序
            if(hasApproveOrder)
            {
                cmList = cmList.stream().sorted(Comparator.comparing(CommentBox::getApproveOrder)).collect(Collectors.toList());
                //重新洗label 重复代码 待优化
                for(int i=0;i< cmList.size();i++)
                {
                    CommentBox cm = cmList.get(i);
                    if(i>=1 && cm.getLabel().equals(cmList.get(i-1).getLabel()))
                    {
                        cm.setHasLabel(false);
                        labelSize ++;
                    }else{
                        cm.setHasLabel(true);
                        cmList.get(lastLabelIndex).setLabelSize(labelSize);
                        labelSize = 1;
                        lastLabelIndex = i;
                    }
                    if(i == cmList.size()-1)
                    {
                        cmList.get(lastLabelIndex).setLabelSize(labelSize);
                    }
                }
            }
            //

        }
        return ResultGenerator.genSuccessResult(cmList);
    }

    /**
     * @Description:：更新/修改步骤意见
     * @param：MyTask
     * @author：CG
     */
    @ApiOperation(value = "updateComment")
    @PostMapping("/updateComment")
    public Result updateComment(@RequestBody MyTask myTask, @RequestParam(required = false) String label) {
        try {
            Auth auth = JwtUtils.GetAuths();
            //意见不为空的时候处理意见相关逻辑判断
            //判断当前步骤与上一步是否为同一个人,若不相同：保存当前意见
            //                             若相同： 当前步骤为承办人审批办理步骤：审批意见跟上次保存是否一致，若一致，则不更新意见
            //                                     当前步骤为领导审批步骤：当前taskid与上次审批taskid是否一致，若不一致，则更新意见
            //                                                                                         若一致，判断意见与上次意见是否一致：若一致，则不更新意见；
            //                                                                                                                       若不一致，则删除上次意见，更新当前意见
            if (StringUtils.isNotEmpty(myTask.getComment())) {
                ActHiComment prevComment = actHiCommentService.getCommentByApproveLoopSettingId(myTask.getApproveLoopSettingId());
                if(prevComment != null && prevComment.getFullMsg() != null)
                {
                    if(prevComment.getUserId().equals(auth.getUserID()))
                    {
                        if(myTask.getIsApproveHandleStep())
                        {
                            if(new String(prevComment.getFullMsg()).equals(myTask.getComment()))
                            {
                                return ResultGenerator.genSuccessResult("保存成功");
                            }
                        }
                        else
                        {
                            if(myTask.getTaskid().equals(prevComment.getTaskId()))
                            {
                                if(new String(prevComment.getFullMsg()).equals(myTask.getComment()))
                                {
                                    return ResultGenerator.genSuccessResult("保存成功");
                                }else{
                                    confActFlowinfoService.deleteComment(myTask.getId());
                                }
                            }
                        }
                    }
                }
                managementService.executeCommand(new MyTaskDto(myTask.getId(), myTask.getProcessInstanceId(), myTask.getComment(), label));
            }
            return ResultGenerator.genSuccessResult("保存成功");
        } catch (Exception e) {
            e.printStackTrace();
            return ResultGenerator.genFailResult("保存失败");
        }
    }

                    if(prevComment.getUserId().equals(auth.getUserID()))
                    {
                        if(myTask.getIsApproveHandleStep())
                        {
                            if(new String(prevComment.getFullMsg()).equals(myTask.getComment()))
                            {
                                return ResultGenerator.genSuccessResult("保存成功");
                            }
                        }
                        else
                        {
                            if(myTask.getTaskid().equals(prevComment.getTaskId()))
                            {
                                if(new String(prevComment.getFullMsg()).equals(myTask.getComment()))
                                {
                                    return ResultGenerator.genSuccessResult("保存成功");
                                }else{
                                    confActFlowinfoService.deleteComment(myTask.getId());
                                }
                            }
                        }
                    }
                }
                managementService.executeCommand(new MyTaskDto(myTask.getId(), myTask.getProcessInstanceId(), myTask.getComment(), label));
            }
            return ResultGenerator.genSuccessResult("保存成功");
        } catch (Exception e) {
            e.printStackTrace();
            return ResultGenerator.genFailResult("保存失败");
        }
    }

    public void setSqlModel(String flowKey, SqlModel sqlModel) {
        if (StringUtils.isNotEmpty(flowKey)) {
            if (flowKey.contains(",")) {
                String orKey = "";
                for (String key : flowKey.split(",")) {
                    if (StringUtils.isNotEmpty(orKey)) {
                        orKey += " or a.flowkey like '%" + key + "%'";

                    } else {
                        orKey = "a.flowkey like '%" + key + "%'";

                    }
                }
                sqlModel.setSearchSql(sqlModel.getSearchSql() + "(" + orKey + ") AND ");

            } else {
                sqlModel.setSearchSql(sqlModel.getSearchSql() + " a.flowkey like " + "'%" + flowKey + "%' AND ");
            }
        }
    }

    @ApiOperation(value = "返回mytask对象")
    @PostMapping("/backMyTask")
    @Secret(value = MyTask.class, encryptStrName = "")
    public Result backMyTask(@RequestParam(required = false) String taskId,@RequestParam(required = false,defaultValue = "") String type) {
        if (StringUtils.isNotEmpty(taskId)) {//办结线索无工作流记录
            Task newtask = taskService.createTaskQuery().taskId(taskId).singleResult();
            MyTask myTask = confActFlowinfoService.backMyTask(taskId);
            //如果当前步骤为退回并且当前存在审批发送步骤
            if(newtask !=null && "back".equals(newtask.getDescription()) && myTask.getSendNodes()!=null && myTask.getSendNodes().size()!=0)
            {
                BusActTask busActTask = busActTaskService.findSendTaskByToTaskId(newtask.getId());
                if(busActTask!=null)
                {
                    Optional<ActTaskApproveLoopsetting> backLoopOpt = myTask.getSendNodes().stream().filter(s->s.getId().equals(busActTask.getApproveLoopSettingId())).findFirst();
                    if(backLoopOpt.isPresent())
                    {

                        try {
                            ActTaskApproveLoopsetting backNode = (ActTaskApproveLoopsetting) backLoopOpt.get().clone();
                            backNode.setApprovenode("发还退回人--" + backNode.getApprovenode());
                            backNode.setApproveNodeType("send-to-back");
                            backNode.setApproveuserid(busActTask.getReceiveUserId());
                            myTask.getSendNodes().add(0, backNode);
                        } catch (CloneNotSupportedException e) {
                            e.printStackTrace();
                        }
                    }
                }
            }
            if(null == myTask){
                return ResultGenerator.genSuccessCodeResult("", 700);
            }
            //查询审批配置
            List<ActTaskApproveLoopsetting> approveLoopSetting = actTaskApproveLoopsettingService.getApproveLoopListByExecutionId(myTask.getExecutionId());
            //查询已经过的审批配置
            List<String> runApproveLoopSetting = actTaskApproveLoopsettingService.getRunApproveLoopListByExecutionId(myTask.getExecutionId());
            if(approveLoopSetting.size()!=0)
            {
                myTask.setCurrentApproveList(approveLoopSetting);
                myTask.setRunApproveLists(runApproveLoopSetting);
                //查询当前审批loopid
                String latestLoopId = busActTaskService.findLatestLoopIdByExecutionId(myTask.getExecutionId());
                myTask.setCurrentApproveloopsettingid(latestLoopId);
            }
            if("1".equals(type)){//前端在办理按钮处理传值后端更改已读状态(查看按钮不传)
                if(myTask.getApproveuser().equals(JwtUtils.GetAuths().getUserID())){//同时查看人id和承办人id相同才会修改已读状态
                    actGeBytearrayService.updateIsRead(taskId);//打开表单标记已读，防止撤回
                }
            }
            MyTask newMyTask = flowableUtil.nextStep(myTask);
            if (null == newtask) {
                return ResultGenerator.genSuccessCodeResult(newMyTask, 700);
            }else {
                return ResultGenerator.genSuccessResult(newMyTask);
            }
        }
        return ResultGenerator.genFailResult("数据有误");
    }

    public static String encode(String str) throws Exception {
        String hexString = "GHEFCDABONMPIJKL";
        //根据编码获取字节数组
        byte[] bytes = str.getBytes("GBK");
        StringBuilder sb = new StringBuilder(bytes.length * 2);
        //将字节数组中每个字节拆解成2位16进制整数
        for (int i = 0; i < bytes.length; i++) {
            sb.append(hexString.charAt((bytes[i] & 0xf0) >> 4));
            sb.append(hexString.charAt((bytes[i] & 0x0f) >> 0));
        }
        String s = sb.toString();
        char[] cs = s.toCharArray();
        for (int j = 0; j < cs.length; j++) {
            if (j % 2 == 0) {
                switch (cs[j]) {
                    case 'M':
                        cs[j] = 'O';
                        break;
                    case 'O':
                        cs[j] = 'M';
                        break;
                    case 'F':
                        cs[j] = 'H';
                        break;
                    case 'H':
                        cs[j] = 'F';
                        break;
                    case 'B':
                        cs[j] = 'D';
                        break;
                    case 'D':
                        cs[j] = 'B';
                        break;
                    case 'A':
                        cs[j] = 'C';
                        break;
                    case 'C':
                        cs[j] = 'A';
                        break;
                    case 'E':
                        cs[j] = 'G';
                        break;
                    case 'G':
                        cs[j] = 'E';
                        break;
                }
            } else {
                switch (cs[j]) {
                    case 'I':
                        cs[j] = 'K';
                        break;
                    case 'K':
                        cs[j] = 'I';
                        break;
                    case 'P':
                        cs[j] = 'N';
                        break;
                    case 'N':
                        cs[j] = 'P';
                        break;
                    case 'L':
                        cs[j] = 'J';
                        break;
                    case 'J':
                        cs[j] = 'L';
                        break;
                }
            }
        }
        return new String(cs);
    }
