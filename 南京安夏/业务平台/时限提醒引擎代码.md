
/**
 * 统一API响应结果封装
 */
public class Result<T> {
    private int code;
    private String message;
    private T data;
    public Result setCode(ResultCode resultCode) {
        this.code = resultCode.code();
        return this;
    }

    public Result setCode(int code) {
        this.code = code;
        return this;
    }

    public int getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }

    public Result setMessage(String message) {
        this.message = message;
        return this;
    }

    public T getData() {
        return data;
    }

    public Result setData(T data) {
        this.data = data;
        return this;
    }

    @Override
    public String toString() {
        return JSON.toJSONString(this);
    }
}
/**
 * 响应码枚举，参考HTTP状态码的语义
 */
public enum ResultCode {
    SUCCESS(200),//成功
    FAIL(400),//失败
    UNAUTHORIZED(401),//未认证（签名错误）
    NOT_FOUND(404),//接口不存在
    INTERNAL_SERVER_ERROR(500);//服务器内部错误


    private final int code;

    ResultCode(int code) {
        this.code = code;
    }

    public int code() {
        return code;
    }
}

/**
 * 时限提醒计算模板
 *
 * @author littlemoneygod
 * @since 2022/12/23 14:18
 */
@Slf4j
@Component
public class TimeRemindCalculation {

    @Autowired
    private StatisticsRemindResultService statisticsRemindResultService;

    @Autowired
    private ActHiProcinstService actHiProcinstService;

    @Autowired
    private ConfRemindVariableService confRemindVariableService;

    @Autowired
    private FlowStatisticsBfTaskService flowStatisticsBfTaskService;

    @Autowired
    private StatisticsHandleSectionService statisticsHandleSectionService;

    @Autowired
    private ConfRemindResultService confRemindResultService;

    @Autowired
    private ConfRemindService confRemindService;

    @Autowired
    private RemindResultHandleService remindResultHandleService;

    @Autowired
    private BaseBusiReBusiFlowkeyService baseBusiReBusiFlowkeyService;

    @Autowired
    private LogStatisticsService logStatisticsService;

    public void execute(String flowBusiKey){
        //查询计算为否的结论重新计算
        List<ConfRemind> confReminds = getCalculateConfRemind(flowBusiKey);
        List<String> remindJsonIds = new ArrayList<>();
        List<StatisticsRemindResult> remindResults = new ArrayList<>();
        for(ConfRemind confRemind : confReminds){
            try {
                //记录重新计算的jsonid
                remindJsonIds.add(confRemind.getId());
                //将时限提醒变量配置（常量+变量）存入map
                Map<String,Object> variables = getVariables(confRemind, flowBusiKey);
                //递归json，获取结论
                String resultId = getResultId(confRemind.getJson(), variables);
                //根据结论计算时限提醒结果
                StatisticsRemindResult statisticsRemindResult = calculateResult(resultId, variables);
                statisticsRemindResult.setFlowBusiKey(flowBusiKey);
                statisticsRemindResult.setRemindJsonId(confRemind.getId());
                remindResults.add(statisticsRemindResult);
            } catch (Exception e) {
                e.printStackTrace();
                logStatisticsService.save(flowBusiKey,confRemind.getId(),null,"remind", ExceptionUtils.getStackTrace(e));
            }
        }
        //删除需要重新计算的结论
        if(remindJsonIds.size() > 0){
            statisticsRemindResultService.delRecalculate(flowBusiKey, remindJsonIds);
        }
        //计算结果入库
        if(remindResults.size()>0){
            statisticsRemindResultService.batchSave(remindResults);
        }
    }

    /**
     * 获取所有变量
     * @author littlemoneygod
     * @date 2022/12/26 11:22
     * @param confRemind
     * @param flowBusiKey
     * @return java.util.Map<java.lang.String,java.lang.Object>
     */
    protected Map<String,Object> getVariables(ConfRemind confRemind, String flowBusiKey){
        Map<String, Object> map = new HashMap<>();
        //获取常态变量
        getStaticData(map, confRemind, flowBusiKey);
        //获取配置变量
        getVariableData(map, confRemind, flowBusiKey);
        return map;
    }

    /**
     * 获取常量变量值
     * @author littlemoneygod
     * @date 2022/12/23 17:16
     * @param map
     * @param confRemind
     * @param flowBusiKey
     * @return void
     */
    protected void getStaticData (Map<String,Object> map, ConfRemind confRemind, String flowBusiKey){
        //当前时间
        map.put(RemindConstant.now.name(), System.currentTimeMillis());
        //登记时间
        map.put(RemindConstant.create.name(), getCreateTime(flowBusiKey));
        //办理时限
        map.put(RemindConstant.due.name(), confRemind.getDueDate() * 24 * 60 * 60 * 1000L);
        //亮灯1
        map.put(RemindConstant.light1.name(), confRemind.getLight1() * 24 * 60 * 60 * 1000L);
        //亮灯2
        map.put(RemindConstant.light2.name(), confRemind.getLight2() * 24 * 60 * 60 * 1000L);
    }

    /**
     * 获取配置变量值
     * @author littlemoneygod
     * @date 2022/12/26 11:22
     * @param map
     * @param confRemind
     * @param flowBusiKey
     * @return void
     */
    protected void getVariableData(Map<String,Object> map, ConfRemind confRemind, String flowBusiKey){
        List<ConfRemindVariable> variables = confRemindVariableService.getConfRemindVariableListsByRemindJsonId(confRemind.getId());
        Map<String, Object> variableMap = new HashMap<>();
        for(ConfRemindVariable variable:variables){
            //todo 待优化为具体类型来判断而不是表名
            if(variable.getTableName().toUpperCase().equals(RemindConstant.STATISTICS_BF_TASK.name())){
                //自定义变量
                variableMap = flowStatisticsBfTaskService.selectMaps(flowBusiKey, variable.getField());
            }else if(variable.getTableName().toUpperCase().equals(RemindConstant.STATISTICS_HANDLE_SECTION.name())){
                //办理情况
                variableMap = statisticsHandleSectionService.queryFieldValueById(variable.getSectionId(), flowBusiKey, variable.getField());
            }
            map.put(variable.getVariableName(), null);
            if(!CollectionUtils.isEmpty(variableMap)){
                if(variableMap.get(variable.getField()) instanceof Date){
                    Date date = (Date) variableMap.get(variable.getField());
                    map.put(variable.getVariableName(), date.getTime());
                }else {
                    map.put(variable.getVariableName(), variableMap.get(variable.getField()));
                }
            }
        }
    }

    /**
     * 获取创建时间
     * @author littlemoneygod
     * @date 2022/12/23 17:08
     * @param flowBusiKey
     * @return long
     */
    protected long getCreateTime(String flowBusiKey){
        LocalDate localDate = actHiProcinstService.getStartTime(flowBusiKey);
        return localDate == null ? null : localDate.atStartOfDay(ZoneOffset.ofHours(8)).toInstant().toEpochMilli();
    }

    /**
     * 根据配置获取最终结论id
     * @author littlemoneygod
     * @date 2022/12/26 11:42
     * @param json
     * @param dataMap
     * @return java.lang.String
     */
    protected String getResultId(String json, Map<String,Object> dataMap){
        RemindJson remindJson = JSONObject.parseObject(json,RemindJson.class);
        RemindJson result = toGetRemindJson(remindJson,dataMap);
        return result.getExpression();
    }

    /**
     * 递归遍历节点返回最终结论
     * @author littlemoneygod
     * @date 2022/12/26 11:43
     * @param remindJson
     * @param dataMap
     * @return com.anshare.calculation.dto.RemindJson
     */
    @SneakyThrows
    private RemindJson toGetRemindJson(RemindJson remindJson, Map<String,Object> dataMap){
        if(false != remindJson.getFlag()){
            String el = remindJson.getExpression();
            if(el.startsWith("${") && el.endsWith("}")){
                Boolean flag = EleUtil.getBoolResult(el,dataMap);
                remindJson = remindJson.getChildren().stream().filter(e->e.getTrend().equals(flag)).collect(Collectors.toList()).get(0);
                return toGetRemindJson(remindJson,dataMap);
            }
        }
        return remindJson;
    }

    /**
     * 根据结论配置计算结果
     * @author littlemoneygod
     * @date 2022/12/26 14:35
     * @param resultId
     * @param variables
     * @return com.anshare.calculation.entity.StatisticsRemindResult
     */
    protected StatisticsRemindResult calculateResult(String resultId, Map<String,Object> variables){
        StatisticsRemindResult statisticsRemindResult = new StatisticsRemindResult();
        statisticsRemindResult.setResultId(resultId);
        statisticsRemindResult.setIsNegate(false);
        ConfRemindResult confRemindResult = confRemindResultService.selectById(resultId);
        if(confRemindResult == null){
            log.error("时限提醒结论配置为空，resultId：" + resultId);
            return statisticsRemindResult;
        }
        //标题
        statisticsRemindResult.setTitle(confRemindResult.getTitle());
        //开始日期
        statisticsRemindResult.setStartTime(calculateDate(confRemindResult.getStartTime(), variables));
        //结束日期
        statisticsRemindResult.setEndTime(calculateDate(confRemindResult.getEndTime(), variables));
        //是否结束
        Boolean isFinished = statisticsRemindResult.getEndTime() != null;
        //已结束则停止计算
        statisticsRemindResult.setIsStopCalc(isFinished);
        //如果还没开始，则不用进行后续计算
        if(statisticsRemindResult.getStartTime() == null){
            return statisticsRemindResult;
        }
        //预计结束日期 starttime(开始日期) + duetime(办理时限)
        statisticsRemindResult.setExpectEndTime( statisticsRemindResult.getStartTime().plusDays((Long) variables.get(RemindConstant.due.name())/1000/24/3600));
        //计算起算日期与预期结束日期时间差
        int intervalDays = remindResultHandleService.circulateOutTime(statisticsRemindResult.getEndTime(), statisticsRemindResult.getExpectEndTime());
        //负数则已超期
        if( intervalDays < 0 ){
            statisticsRemindResult.setIsOutTime(true);
            //如果已结束，计算超期日期
            if(isFinished){
                statisticsRemindResult.setOutTime(Math.abs(intervalDays));
            }
        }
        return statisticsRemindResult;
    }

    /**
     * el表达式计算日期
     * @author littlemoneygod
     * @date 2022/12/26 17:16
     * @param expression
     * @param variables
     * @return java.time.LocalDate
     */
    private LocalDate calculateDate(String expression, Map<String,Object> variables){
        String startTime = EleUtil.getStrResult(expression, variables);
        return StringUtils.isNotEmpty(startTime) ? DateUtils.timestamp2LocalDate(Long.parseLong(startTime)) : null;
    }

     * @author littlemoneygod
     * @date 2022/5/17 10:32
     * @param targetClass
     * @param funcName
     * @return com.baomidou.mybatisplus.core.toolkit.support.SFunction
     */
    public static SFunction genSFunction(Class targetClass, Class fieldClass, String funcName) throws Throwable {
        //创建查找句柄
        MethodHandles.Lookup lookup = MethodHandles.lookup();
        //方法类型，描述返回值和参数，参数1：返回值类型，参数2及之后：被调用方法的参数类型
        MethodType methodType = MethodType.methodType(fieldClass, targetClass);
        //查找方法句柄，参数1：调用类，参数2：方法名，参数3：方法类型
        MethodHandle handle = lookup.findVirtual(targetClass, funcName, MethodType.methodType(fieldClass));
        final CallSite site;
        site = LambdaMetafactory.altMetafactory(lookup,
                "invoke",
                MethodType.methodType(SFunction.class),
                methodType,
                handle,
                methodType, 1);
        SFunction sfunc = (SFunction) site.getTarget().invokeExact();
        return sfunc;

     * @author littlemoneygod
     * @date 2022/5/18 00:14
     * @param clazz
     * @return java.util.List<java.lang.reflect.Field>
     */
    public static List<Field> getAllField(Class clazz) throws NoSuchFieldException {
        List<Field> fieldList = new ArrayList<>() ;
        while (clazz != null) {
            fieldList.addAll(Arrays.asList(clazz.getDeclaredFields()));
            clazz = clazz.getSuperclass();
        }
        return fieldList;
    }

    /**
     * 获取类与父类匹配的field字段
     * @author littlemoneygod
     * @date 2022/5/18 00:14
     * @param clazz
     * @return java.util.List<java.lang.reflect.Field>
     */
    public static Field getField(Class clazz, String fieldName) {
        while (clazz != null) {
            Field[] fields = clazz.getDeclaredFields();
            for(Field field : fields){
                if(field.getName().equals(fieldName)){
                    return field;
                }
            }
            clazz = clazz.getSuperclass();
        }
        return null;
    }

    /**
     * 判断对象所有属性是否为 null
     *
     * @param obj 要判断的对象
     * @return 如果所有属性都为 null，则返回 true；否则返回 false。
     */
    public static boolean isAllFieldsNull(Object obj) throws IllegalAccessException, NoSuchFieldException {
        List<Field> fields = getAllField(obj.getClass());
        for (Field field : fields) {
            if (!field.getName().equals("serialVersionUID")) {
                field.setAccessible(true);
                if (field.get(obj) != null) {
                    return false;
                }
            }
        }
        return true;
    }

        int days = DateUtils.daysBetweenLocaldate(LocalDate.now(), result.getExpectEndTime());
        //当前时间 < 超期时间，则已超期
        if(days < 0){
            color = RemindConstant.due.name();
            days = Math.abs(days);
        }else{
            //当前时间 < 亮灯实现1
            if(days <= confRemind.getLight1()){
                color = RemindConstant.light1.name();
            }
            //当前时间 < 亮灯实现2
            else if(days <= confRemind.getLight2()){
                color = RemindConstant.light2.name();
            }
        }
        map.put("overDueMsg", confRemind.getType() + "("  + days + ")");
        map.put("remindType", confRemind.getType());
        map.put("remindDays", days+"");
        map.put("lightMsg", color);
        return map;
    }

    @Override
    public RemindResultDto getSingleRemindDetail(String remindResultId){
        //查询时限提醒计算结果
        StatisticsRemindResult statisticsRemindResult = statisticsRemindResultService.selectById(remindResultId);
        CirculateEnum.REMIND_RESULT_NOT_NULL.assertNotNull(statisticsRemindResult);
        //查询时限提醒配置
        ConfRemind confRemind = confRemindService.selectById(statisticsRemindResult.getRemindJsonId());
        CirculateEnum.CONF_REMIND_NOT_NULL.assertNotNull(confRemind);
        //赋值
        RemindResultDto remindResultDto = new RemindResultDto();
        remindResultDto.setRemindResultId(statisticsRemindResult.getId());
        remindResultDto.setRemindJsonId(statisticsRemindResult.getRemindJsonId());
        remindResultDto.setDescription(confRemind.getDescription());
        remindResultDto.setName(confRemind.getName());
        remindResultDto.setStartTime(statisticsRemindResult.getStartTime());
        remindResultDto.setExpectEndTime(statisticsRemindResult.getExpectEndTime());
        remindResultDto.setEndTime(statisticsRemindResult.getEndTime());
        //计算起算日期与预期结束日期时间差
        int intervalDays = circulateOutTime(statisticsRemindResult.getEndTime(), statisticsRemindResult.getExpectEndTime());
        //负数则已超期
        if( intervalDays < 0 ){
            remindResultDto.setIsOutTime(true);
            //计算超期日期
            remindResultDto.setOutTime(Math.abs(intervalDays));
        }else{
            remindResultDto.setIsOutTime(false);
            //ji算剩余日期
            remindResultDto.setLeftTime(intervalDays);
         }
        return remindResultDto;
    }

    @Override
    public int circulateOutTime(LocalDate endTime, LocalDate expectEndTime){
        //计算起始日期:已结束->结束日期；未结束->当前日期
        LocalDate intervalStartDate = endTime != null ? endTime : LocalDate.now();
        //计算起始日期与预期结束日期差值，正数则未超期，负数则已超期
        int intervalDays = DateUtils.daysBetweenLocaldate(intervalStartDate, expectEndTime);
        return intervalDays;
    }

    @Override
    public WaitTimeRemindNumber getWaitTimeRemindNumber(List<String> flowBusiKeys) {
        int due = 0;
        int light1 = 0;
        int light2 = 0;
        //查询时限提醒结果
        Map<String, Map<String, String>> remindResultMap = getRemindDetail(flowBusiKeys);
        for(String key : remindResultMap.keySet()){
            switch (remindResultMap.get(key).get("lightMsg")){
                case "due":
                    due++;
                    break;
                case "light1":
                    light1++;
                    break;
                case "light2":
                    light2++;
                    break;
            }
        }
        WaitTimeRemindNumber waitTimeRemindNumber = new WaitTimeRemindNumber();
        waitTimeRemindNumber.setOvertime(due);
        waitTimeRemindNumber.setLight1(light1);
        waitTimeRemindNumber.setLight2(light2);
        return waitTimeRemindNumber;
    }

    /**

     * @author littlemoneygod
     * @date 2022/12/21 14:02
     * @param confActHandleSection
     * @param flowBusiKey
     * @return com.anshare.calculation.entity.StatisticsHandleSection
     */
    protected StatisticsHandleSection getHandleSection(ConfActHandleSection confActHandleSection, String flowBusiKey){
        StatisticsHandleSection section = new StatisticsHandleSection();
        section.setSectionId(confActHandleSection.getId());
        section.setFlowBusiKey(flowBusiKey);
        String taskId = null;
        //开始时间判断
        CirculateDateDto startDateDto = getDate(confActHandleSection.getStartJson(), confActHandleSection.getFlowkey(), flowBusiKey);
        if(startDateDto != null){
            section.setStartTime(startDateDto.getDate());
            taskId = startDateDto.getTaskId();
        }
        //结束时间判断
        CirculateDateDto endDateDto = getDate(confActHandleSection.getEndJson(), confActHandleSection.getFlowkey(), flowBusiKey);
        if(endDateDto != null){
            section.setEndTime(endDateDto.getDate());
        }
        //办理信息获取
        setHandleMsg(section, confActHandleSection.getDeptType(), taskId, flowBusiKey);
        //todo自定义业务信息处理
        return section;
    }

    /**
     * 根据步骤类型获取时间
     * @author littlemoneygod
     * @date 2022/12/20 20:49
     * @param json
     * @param flowKey
     * @param flowBusiKey
     * @return java.time.LocalDate
     */
    private CirculateDateDto getDate(String json, String flowKey, String flowBusiKey){
        if(StringUtils.isBlank(json)){
            return null;
        }
        HandleSectionDto handleSectionDto = JSONObject.parseObject(json, HandleSectionDto.class);
        // 判断该时间是 1:节点判断 2:字段抓取
        if(null == handleSectionDto.getDateType()){
            return null;
        }
        if(handleSectionDto.getDateType() == 1){
            return getDateByFlowElement(handleSectionDto.getFlowElementIdJudge(), handleSectionDto.getFlowElementIds(), flowBusiKey);
        }else{
            return getDateByField(handleSectionDto.getFieldJudge(), handleSectionDto.getFlowElementIds(), flowKey, flowBusiKey);
        }
    }

    /**
     * 根据配置类型获取计时间
     * @author littlemoneygod
     * @date 2022/12/20 20:49
     * @param flowElementIdJudge
     * @param flowElementIds
     * @param flowBusiKey
     * @return java.time.LocalDate
     */
    private CirculateDateDto getDateByFlowElement(HandleSectionDto.FlowElementIdJudge flowElementIdJudge, String[] flowElementIds, String flowBusiKey){
        CirculateDateDto dto = null;
        boolean isLatest = flowElementIdJudge.getFlowElementIdDateSelectType() == 1;
        switch(flowElementIdJudge.getFlowElementIdDateType()){
            //进入时间
            case 1:
                dto = actHiTaskinstService.getCirculateDate(flowBusiKey, flowElementIds, isLatest, true);
                break;
            //发走时间
            case 2:
                dto = actHiTaskinstService.getCirculateDate(flowBusiKey, flowElementIds, isLatest, false);
                break;
            //审批通过时间
            case 3:
                dto = busiFormService.getApproveDate(flowBusiKey, flowElementIds, isLatest);
                break;
            //办结时间
            case 4:
                LocalDate localDate = actHiProcinstService.getEndTime(flowBusiKey);
                dto = CirculateDateDto.builder()
                        .date(localDate)
                        .build();
                break;
        }
        return dto;
    }

    /**
     * 根据字段获取时间
     * todo feign处理
     * @author littlemoneygod
     * @date 2022/12/20 20:49
     * @param fieldJudge
     * @return java.time.LocalDate
     */
    private CirculateDateDto getDateByField(HandleSectionDto.FieldJudge fieldJudge,  String[] flowElementIds, String flowKey, String flowBusiKey){
        InstanceExecute execute = InstanceExecuteFactory.getInstance(flowKey);
        Map<String, String> instMap = execute.getBusiInstMap(flowBusiKey);
        if(CollectionUtils.isEmpty(instMap)){
            return null;
        }
        if(fieldJudge == null || fieldJudge.getFieldDate() == null || fieldJudge.getFieldDate().length != 2){
            return null;
        }
        String field = fieldJudge.getFieldDate()[1];
        //取字段值
        Map<String, Object> map = bizFieldUtil.getField(fieldJudge.getFieldDate()[0], Arrays.asList(field), flowBusiKey, instMap.entrySet().stream().findFirst().get().getKey());
        if(CollectionUtils.isEmpty(map)){
            return null;
        }
        LocalDate circulateDate = null;
        if(map.get(field) instanceof Timestamp){
            Timestamp exTime = (Timestamp)map.get(field);
            circulateDate = exTime.toLocalDateTime().toLocalDate();
        }else if(map.get(field) instanceof java.sql.Date){
            java.sql.Date date = (Date) map.get(field);
            circulateDate = date.toLocalDate();
        }
        CirculateDateDto circulateDateDto = CirculateDateDto.builder()
                .date(circulateDate)
                .build();
        if(null != flowElementIds && flowElementIds.length > 0){
            CirculateDateDto date = actHiTaskinstService.getCirculateDate(flowBusiKey, flowElementIds, false, true);
            circulateDateDto.setTaskId( null != date ? date.getTaskId() : "");
        }

        return circulateDateDto;
    }

     * @date 2022/12/21 10:44
     * @param section
     * @param deptType
     * @param taskId
     * @param flowBusiKey
     * @return void
     */
    private void setHandleMsg(StatisticsHandleSection section, int deptType, String taskId, String flowBusiKey){
        switch(deptType){
            //创建人
            case 0:
                BaseBusiReBusiFlowkey flowkey = baseBusiReBusiFlowkeyService.selectById(flowBusiKey);
                if(flowkey != null){
                    section.setHandleUserId(flowkey.getCreateUserId());
                    section.setHandleUserName(flowkey.getCreateUserName());
                    section.setHandleDeptId(flowkey.getCreateDeptId());
                    section.setHandleDeptName(flowkey.getCreateDeptName());
                }
                break;
            //承办部门
            case 1:
                if(StringUtils.isNotBlank(taskId)){
                    BusiForm busiForm = busiFormService.getByTaskId(taskId);
                    if(busiForm!=null){
                        section.setHandleUserId(busiForm.getSubmitStaffId());
                        section.setHandleUserName(busiForm.getSubmitStaffName());
                        section.setHandleDeptId(busiForm.getSubmitDeptId());
                        section.setHandleDeptName(busiForm.getSubmitDeptName());
                    }
                }
                break;
            //当前部门
            case 2:
                if(StringUtils.isNotBlank(taskId)){
                    BusiActTask task = busiActTaskService.selectById(taskId);
                    if(task!=null){
                        section.setHandleUserId(task.getReceiveUserId());
                        section.setHandleUserName(task.getReceiveUserName());
                        section.setHandleDeptId(task.getReceiveDeptId());
                        CommDept commDept = commDeptService.selectById(task.getReceiveDeptId());
                        if(commDept!=null){
                            section.setHandleDeptName(commDept.getName());
                        }
                    }
                }
                break;
        }
    }

     * @return null

        if(StringUtils.isBlank(confActHandleSection.getFieldJson())){

        }
        HandleSectionFieldDto handleSectionFieldDto = JSONObject.parseObject(confActHandleSection.getFieldJson(), HandleSectionFieldDto.class);
        if(StringUtils.isNotBlank(handleSectionFieldDto.getFieldDate())){
            String[] tableColumn = handleSectionFieldDto.getFieldDate().split(",");
            if(tableColumn.length != 2){
                return;
            }
            String tableName = tableColumn[0];
            String column = tableColumn[1];
            //创建更新实体对象
            String key =  CaseFormat.LOWER_UNDERSCORE.to(CaseFormat.UPPER_CAMEL, tableName.toLowerCase());
            Class<?> targetClass = null;
            try {
                targetClass = Class.forName(packageConstant.getEntityPkg() + key);
            } catch (ClassNotFoundException e) {
                return;
            }
            MyBaseMapper<Object> mapper = beanFactory.getBean(com.anshare.base.util.StringUtils.firstCharToLowCase(key) + packageConstant.getMapperSuffix(), MyBaseMapper.class);
            //查询表主键外键关系
            ConfFlowBusiEr confFlowBusiEr = confFlowBusiErService.getConfFlowBusiEr(tableName);
            if(confFlowBusiEr == null || com.anshare.base.util.StringUtils.isBlank(confFlowBusiEr.getMultiSourceType())){
                return;
            }
            //与flowbusikey一对一，则主键为flowbusikey
            if(confFlowBusiEr.getType() == 1){
                List<Map<String, Object>> result  =  queryList(targetClass, column, flowBusiKey, mapper);
                updateField(targetClass, section, handleSectionFieldDto, column, flowBusiKey, mapper, result);
            }else{
                //查询主键
                InstanceExecute execute = InstanceExecuteFactory.getInstance(confActHandleSection.getFlowkey());
                Map<String, String> objectMap = execute.getBusiInstMap(flowBusiKey);
                for(String id : objectMap.keySet()){
                    List<Map<String, Object>> result  =  queryList(targetClass, column, id, mapper);
                    updateField(targetClass, section, handleSectionFieldDto, column, id, mapper, result);
                }
            }
        }
    }


    /**
     * 查询该字段是否有值
     * @author zhp
     * @date 2023-02-21 19:19
     * @param targetClass
     * @param column
     * @param id
     * @param mapper
     * @return java.util.List<java.util.Map<java.lang.String,java.lang.Object>>
     */
    private List<Map<String, Object>> queryList(Class<?> targetClass, String column, String id,MyBaseMapper<Object> mapper){
        //查询该字段有没有值
        Wrapper<Object> queryWrapper = null;
        try {
            queryWrapper = Wrappers.query()
                    .setEntity(targetClass.newInstance())
                    .select(column)
                    .eq(BusiConstant.PrimaryKey,id);
        } catch (Exception e) {
            e.printStackTrace();
            log.error("查询数据信息失败，请检查业务表、字段、业务实例是否存在:" + targetClass);
        }
        List<Map<String, Object>> list = mapper.selectMaps(queryWrapper);
        return list;
    }

    /**
     * 根据字段更新
     * @author zhp
     * @date 2023-02-21 19:19
     * @param targetClass
     * @param section
     * @param handleSectionFieldDto
     * @param column
     * @param id
     * @param mapper
     * @param result
     * @return void
     */
    private  void  updateField(Class<?> targetClass, StatisticsHandleSection section, HandleSectionFieldDto handleSectionFieldDto, String column, String id,MyBaseMapper<Object> mapper,List<Map<String, Object>> result){
        LocalDate time = null;
        if(handleSectionFieldDto.getDateType() == 1){
            time = section.getStartTime();
        }else {
            time = section.getEndTime();
        }
        //一直更新或者没有值的时候
        if((handleSectionFieldDto.getIsKeepUpdate()!=null && handleSectionFieldDto.getIsKeepUpdate())|| (CollectionUtils.isEmpty(result)||null == result.get(0))){
            UpdateWrapper<Object> wrapper = null;
            try {
                wrapper = Wrappers.update(targetClass.newInstance());
            } catch (Exception e) {
                e.printStackTrace();
                log.error("查询数据信息失败，请检查业务表、业务实例是否存在：" + targetClass);
            }
            wrapper.set( column,time);
            wrapper.eq(BusiConstant.PrimaryKey,id);
            mapper.update(null, wrapper);
        }
    }
}

/**
 * 维度计算模板
 *
 * @author littlemoneygod
 * @since 2023/2/13 16:15
 */
@Slf4j
@Component
public class DimensionCalculation {

    @Autowired
    private StatisticsDimensionResultService statisticsDimensionResultService;

    @Autowired
    private StatisticsDimensionResultMapper statisticsDimensionResultMapper;

    @Autowired
    private DimensionResultHandleService dimensionResultHandleService;

    @Autowired
    private BusiReFlowBusiService busiReFlowBusiService;

    @Autowired
    private LogStatisticsService logStatisticsService;

    @Autowired
    private ConfCalculateVerifyRuleService confCalculateVerifyRuleService;

    @Autowired
    private BizFieldUtil bizFieldUtil;

    @Autowired
    private CalculateVerifyErrorLogService calculateVerifyErrorLogService;

    private static Logger statisticsLog = LoggerFactory.getLogger("statisticsLog");


    public void execute(DimensionDto dimensionDto){
        //获取流程下所有业务数据
//        setInstIds(dimensionDto);
        //删除所有该件已删除业务数据的统计
        delBusiResult(dimensionDto.getFlowBusiKey(), dimensionDto.getInstIds());
        //校验数据是否通过，不通过的不参与统计
        dimensionDto.setInstIds(verify(dimensionDto.getInstIds(), dimensionDto.getFlowBusiKey()));
        //若校验后没有数据，则不再进行统计
        if(CollectionUtils.isEmpty(dimensionDto.getInstIds())){
            return;
        }
        //获取当前流程所有可用维度
        List<ConfDimension> confDimensions = statisticsDimensionResultService.getByCalculation(dimensionDto);
        //遍历计算维度，获取计算结果
        List<StatisticsDimensionResult> results = calculate(confDimensions, dimensionDto);
        //入库
        saveResults(results);
    }

    /**
     * 设置业务信息
     * @author littlemoneygod
     * @date 2024/11/15 17:35
     * @param dimensionDto
     * @return void
    */
    private void setInstIds(DimensionDto dimensionDto){
        List<BusiReFlowBusi> busiReFlowBusiList = busiReFlowBusiService.getAllBusiReFlowBusi(dimensionDto.getFlowBusiKey());
        FlowBusiEnum.BUSI_RE_FLOW_IS_EMPTY.listNotEmpty(busiReFlowBusiList);
        dimensionDto.setInstIds(busiReFlowBusiList.stream().map(b->b.getInstId()).collect(Collectors.toList()));
        dimensionDto.setActiveInstIds(busiReFlowBusiList.stream().filter(b->b.getIsActive()).map(b->b.getInstId()).collect(Collectors.toList()));
    }

    /**
     * 遍历计算维度
     * @author littlemoneygod
     * @date 2024/11/15 17:07
     * @param confDimensions
     * @param dimensionDto
     * @return java.util.List<com.anshare.calculation.entity.StatisticsDimensionResult>
    */
    private List<StatisticsDimensionResult> calculate(List<ConfDimension> confDimensions, DimensionDto dimensionDto){
        List<StatisticsDimensionResult> results = new ArrayList<>();
        for(ConfDimension confDimension : confDimensions){
            try {
                List<StatisticsDimensionResult> cur = dimensionResultHandleService.calculateResult(confDimension, dimensionDto);
                if(cur.size() > 0){
                    results.addAll(cur);
                }
                statisticsLog.info("维度：" + confDimension.getDimensionName() + "结果：" + cur.size());
            } catch (Exception e) {
                e.printStackTrace();
                logStatisticsService.save(dimensionDto.getFlowBusiKey(),confDimension.getId(), null,"statistics", ExceptionUtils.getStackTrace(e));
            }
        }
        return results;
    }

    /**
     * 保存计算结果
     * @author littlemoneygod
     * @date 2024/11/15 17:07
     * @param results
     * @return void
    */
    private void saveResults(List<StatisticsDimensionResult> results){
        if(results.size() > 0){
            statisticsDimensionResultService.batchSave(results);
        }
    }

    /**
     * 删除该流程下所有业务相关统计数据，便于后续重新计算
     * 此处删除2部分数据：
     * 第一部分：flowbusikey为当前流程的所有数据
     * 第二部分：instid为当前业务的所有数据，此处可能包含flowbusikey不为当前流程的数据
     * @author littlemoneygod
     * @date 2024/11/15 16:41
     * @param flowBusiKey
     * @param instIds
     * @return void
    */
    public void delBusiResult(String flowBusiKey, List<String> instIds){
        LambdaQueryWrapper<StatisticsDimensionResult> wrapper = new LambdaQueryWrapper();
        wrapper.eq(StatisticsDimensionResult::getFlowBusiKey, flowBusiKey).or().in(StatisticsDimensionResult::getInstId, instIds);
        statisticsDimensionResultMapper.delete(wrapper);
    }

    public List<String> verify(List<String> instIds, String flowBusiKey){
        Map<String, Set<String>> rules = confCalculateVerifyRuleService.selectPropertyMap();
        List<String> calInstIds = new ArrayList<>();
        for(String instId : instIds){
            Map<String,Object> param = getDataVal(instId, flowBusiKey, rules);
            //存储异常日志
            List<ConfCalculateVerifyRule> errorRules = verifyResult(param);
            //更新异常数据日志
            addErrorLog(instId, errorRules);
            if(CollectionUtils.isEmpty(errorRules)){
                calInstIds.add(instId);
            }
        }
        return calInstIds;
    }

    public  Map<String,Object> getDataVal(String instId, String flowBusiKey, Map<String, Set<String>> rules){
        Map<String,Object> param = new HashMap<>();
        for(String key : rules.keySet()){
            Map<String, Object> values = bizFieldUtil.getFieldNoEpmty(key, new ArrayList<>(rules.get(key)), flowBusiKey, instId);
            if(CollectionUtils.isEmpty(values)){
                putDataNull(param,key,new ArrayList<>(rules.get(key)));
                continue;
            }
            for(String field : values.keySet()){
                param.put(String.join("_", StringUtils.firstCharToUpCase(StringUtils.toCamelCase(key)), StringUtils.toCamelCase(field)), values.get(field));
            }
        }
        return param;
    }

    private void putDataNull(Map<String, Object> param, String key, ArrayList<String> fields) {
        for(String field : fields){
            param.put(String.join("_", StringUtils.firstCharToUpCase(StringUtils.toCamelCase(key)), StringUtils.toCamelCase(field)), null);
        }
    }

    public List<ConfCalculateVerifyRule> verifyResult(Map<String, Object> dataMap){
        List<ConfCalculateVerifyRule> errorRule = new ArrayList<>();
        List<ConfCalculateVerifyRule> rules = confCalculateVerifyRuleService.selectRules();
        for(ConfCalculateVerifyRule rule : rules){
            try {
                Boolean flag = EleUtil.getBoolResult(rule.getRule(),dataMap);
                //满足条件即报错
                if(flag){
                    errorRule.add(rule);
                }
            } catch (Exception e){
                log.info("规则配置有问题：---"+rule.getRule());
            }
        }
        return errorRule;
    }

    private void addErrorLog(String instId, List<ConfCalculateVerifyRule> errorRules){
        calculateVerifyErrorLogService.delByInstId(instId);
        if(CollectionUtils.isEmpty(errorRules)){
            return;
        }
        List<CalculateVerifyErrorLog> logs = new ArrayList<>();
        for(ConfCalculateVerifyRule rule : errorRules){
            CalculateVerifyErrorLog errorLog = new CalculateVerifyErrorLog();
            errorLog.setInstId(instId);
            errorLog.setRuleId(rule.getId());
            errorLog.setRemark(rule.getRemark());
            logs.add(errorLog);
        }
        calculateVerifyErrorLogService.batchSave(logs);
    }
}
