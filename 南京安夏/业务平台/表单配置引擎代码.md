/**
 * 统一API响应结果封装
 */
public class Result<T> {
    private int code;
    private String message;
    private T data;
    public Result setCode(ResultCode resultCode) {
        this.code = resultCode.code();
        return this;
    }

    public Result setCode(int code) {
        this.code = code;
        return this;
    }

    public int getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }

    public Result setMessage(String message) {
        this.message = message;
        return this;
    }

    public T getData() {
        return data;
    }

    public Result setData(T data) {
        this.data = data;
        return this;
    }

    @Override
    public String toString() {
        return JSON.toJSONString(this);
    }
}
/**
 * 响应码枚举，参考HTTP状态码的语义
 */
public enum ResultCode {
    SUCCESS(200),//成功
    FAIL(400),//失败
    UNAUTHORIZED(401),//未认证（签名错误）
    NOT_FOUND(404),//接口不存在
    INTERNAL_SERVER_ERROR(500);//服务器内部错误


    private final int code;

    ResultCode(int code) {
        this.code = code;
    }

    public int code() {
        return code;
    }
}

    //型如XXX_YYY_ZZZ的子串改为xxxYyyZxx的子串
    public static String slashToFirstLetterUpper(String input) {
        String spliter = "_";
        StringBuffer output = new StringBuffer();
        String[] words = input.toLowerCase().split(spliter);
        for (int i = 0; i < words.length; i++) {
            if (i != 0) {
                output.append(fistLetterToUpper(words[i]));
            }
            else {
                output.append(words[i]);
            }
        }

        return output.toString();
    }

    private static String fistLetterToUpper(String input) {
        if (input == null)
            return "";
        if (input.length() <= 0)
            return "";

        return input.substring(0, 1).toUpperCase() + input.substring(1);
    }

        Condition condition = new Condition(ConfForm.class);
        condition.createCriteria()
                .andEqualTo("tableName", tablename).andEqualTo("isdeleted", 0);
        ConfForm confForm = new ConfForm();
        if(redisUtil.hasKey("confform"+tablename,2)){
            confForm = (ConfForm)redisUtil.get("confform"+tablename,2);
        }else {
            List<ConfForm> list = confFormService.findByCondition(condition);
            if(list.size()==0){
                return ResultGenerator.genFailCodeResult(500,tablename);
            }
            confForm = list.get(0);
            redisUtil.set("form"+tablename,list.get(0),2);
        }
        return ResultGenerator.genSuccessResult(confForm);
    }

    @ApiOperation(value = "dataExport")
    @PostMapping("/dataExport")
    @MyLog(value = "导出记录")
    public void dataExport(@RequestBody @Validated ConditionsModel conditionsModel, HttpServletResponse response) {
        PageHelper.startPage(conditionsModel.getPageIndex(), conditionsModel.getPageSize());
        Condition condition = new Condition(ConfForm.class);
        List<ConfForm> list = confFormService.findByCondition(ConditionUtils.getConditions(condition,conditionsModel.getSearchCondition(),conditionsModel.getOrderCondition()));
        for (int i = 0; i < list.size(); i++){
            ConfForm confForm = list.get(i);
            String formJson = confForm.getFormJson();
            if (formJson.length() > 32767){
                formJson = formJson.substring(0,32765);
                confForm.setFormJson(formJson);
            }
        }
        PageInfo pageInfo = new PageInfo(list);
        ExportExcel.exportExcelPlus(conditionsModel.getExportCondition(),pageInfo.getList(),response);
    }

    @ApiOperation(value = "dataImport")

    public Result dataImport(HttpServletRequest request, MultipartFile file) {
        List<Object> list = ImportExcel.dataInport(request, file , ConfForm.class);
        for (Object object : list) {
            if (object instanceof ConfForm) {
                ConfForm confForm = (ConfForm) object;
                confFormService.save(confForm, true);
            }
        }
        return ResultGenerator.genSuccessResult("成功导入" + list.size() + "条数据");
    }

@Service
public class ConfAdCodelistServiceImpl extends BaseServiceImpl<ConfAdCodelist> implements ConfAdCodelistService {
    @Autowired
    private ConfAdCodelistMapper confAdCodelistMapper;

    @Autowired
    private ConfAdCodelistService confAdCodelistService;

    @Override
    @Cacheable(value = "confAdCodelist",key = "#id")
    public ConfAdCodelist selectById(String id) {
        ConfAdCodelist confAdCodelist = confAdCodelistMapper.selectById(id);
        return confAdCodelist;
    }

    @Override
    @Cacheable(value = "confAdCodelist",key = "#codeType")
    public List<ConfAdCodelist> getByCodeType(String codeType) {
        QueryWrapper wrapper = new QueryWrapper();
        wrapper.eq("code_type",codeType);
        wrapper.orderByAsc("code_order");
        List<ConfAdCodelist> list = confAdCodelistMapper.selectList(wrapper);
        return list;
    }

    @Override
    public List<ConfAdCodelist> getByCodeTypes(String[] codeTypes) {
        List<ConfAdCodelist> list = new ArrayList<>();
        for(String type:codeTypes){
            list.addAll(confAdCodelistService.getByCodeType(type));
        }
        return list;
    }

    @Override
    @Cacheable(value = "confAdCodelist",key = "#codeName")
    public List<ConfAdCodelist> getByCodeName(String codeName) {
        QueryWrapper wrapper = new QueryWrapper();
        wrapper.eq("code_name",codeName);
        wrapper.orderByAsc("code_order");
        List<ConfAdCodelist> list = confAdCodelistMapper.selectList(wrapper);
        return list;
    }

    @Override
    @Cacheable(value = "confAdCodelistDto",key = "#codeType")
    public List<DictDto> getDictDtoByCodeType(String codeType) {
        List<ConfAdCodelist> list = confAdCodelistService.getByCodeType(codeType);
        List<DictDto> dictDtos = list.stream().map(d->new DictDto(d)).collect(Collectors.toList());
        return dictDtos;
    }

    @Override
    public Map<String,List<DictDto>> getByCodeTypes(List<String> codeType) {
        Map<String,List<DictDto>> map = new HashMap<>();
        //遍历查询便于清理缓存
        for(String type : codeType){
            List<DictDto> dictDtos = confAdCodelistService.getDictDtoByCodeType(type);
            map.put(type,dictDtos);
        }
        return map;
    }

    @Override
    @Caching(
            evict={
                    @CacheEvict(value = "confAdCodelist", key = "#confAdCodelist.codeType"),
                    @CacheEvict(value = "confAdCodelistDto", key = "#confAdCodelist.codeType")
            })

    {
        return super.save(confAdCodelist);
    }

            evict={ @CacheEvict(value = "confAdCodelist", key = "#confAdCodelist.id"),
                    @CacheEvict(value = "confAdCodelist", key = "#confAdCodelist.codeType"),
                    @CacheEvict(value = "confAdCodelistDto", key = "#confAdCodelist.codeType")
            })

        return confAdCodelistMapper.updateById(confAdCodelist);

                    @CacheEvict(value = "confAdCodelistDto", key = "#confAdCodelist.codeType")

        return confAdCodelistMapper.deleteById(confAdCodelist.getId());
    }

    @Override
    public ConfAdCodelist getByCodeVal(String val, String type) {
        List<ConfAdCodelist> confAdCodelists = confAdCodelistService.getByCodeType(type);
        if(!CollectionUtil.isEmpty(confAdCodelists)){
            Optional<ConfAdCodelist> confAdCodelist = confAdCodelists.stream().filter(c->val.equals(c.getCodeValue())).findFirst();
            if(confAdCodelist.isPresent()){
                return confAdCodelist.get();
            }
        }
        return null;
    }

@Component

    @Resource
    RedisConfig redisConfig;


    /**
     * 清除指定前缀的缓存
     *
     */
    public void delKeyPre(String messageKeyPre){
        Set<String> keys = redisConfig.getRedisTemplate().execute((RedisCallback<Set<String>>) connection -> {
            Set<String> keysTmp = new HashSet<>();
            Cursor<byte[]> cursor = connection.scan(new ScanOptions.ScanOptionsBuilder().match("*" + messageKeyPre + "*").count(1000).build());
            while (cursor.hasNext()) {
                keysTmp.add(new String(cursor.next()));
            }
            return keysTmp;
        });
        Iterator<String> it = keys.iterator();
        while(it.hasNext()){
            String keyStr = it.next();
            System.out.println(keyStr);
            redisConfig.getRedisTemplate().delete(keyStr);
        }
    }


    /**
     * 设置Str缓存
     * @param key
     * @param val
     */
    public void setStr(String key, String val){
        redisConfig.getRedisTemplate().opsForValue().set(ProjectConstant.REDISKEY+key,val);
    }

    /**
     * 设置Str缓存,带时间（单位毫秒）
     * @param key
     * @param val
     */
    public void setStr(String key, String val, long timeout){
        redisConfig.getRedisTemplate().opsForValue().set(ProjectConstant.REDISKEY+key,val, timeout, TimeUnit.SECONDS);
    }

    /**
     * 设置Str缓存过期时间
     * @param key
     * @param timeout
     */
    public void expire(String key, long timeout){
        redisConfig.getRedisTemplate().expire(key, timeout, TimeUnit.SECONDS);
    }


    /**
     * 判断key是否存在
     * @param key 键
     * @return true 存在 false不存在
     */
    public boolean hasKey(String key,Integer num) {
        try {
            return redisConfig.getRedisTemplateByDb(num).hasKey(ProjectConstant.REDISKEY+key);
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }
    /**
     * 删除缓存
     * @param key 可以传一个值 或多个
     */
    @SuppressWarnings("unchecked")
    public void del(Integer num,String... key) {
        if (key != null && key.length > 0) {
            if (key.length == 1) {
                redisConfig.getRedisTemplateByDb(num).delete(ProjectConstant.REDISKEY+key[0]);
            } else {
                redisConfig.getRedisTemplateByDb(num).delete(CollectionUtils.arrayToList(ProjectConstant.REDISKEY+key));
            }
        }
    }


    /**
    * Created by Anshare on 2021/10/26
    * @author: Julian
    * @description: 删除指定key
    */
    @SuppressWarnings("unchecked")
    public void delWithKey(Integer num,String... key) {
        if (key != null && key.length > 0) {
            if (key.length == 1) {
                redisConfig.getRedisTemplateByDb(num).delete(key[0]);
            } else {
                redisConfig.getRedisTemplateByDb(num).delete(CollectionUtils.arrayToList(key));
            }
        }
    }

    */
    public void delKeyWithPre(Integer num,String messageKeyPre){
        Set<String> keys = redisConfig.getRedisTemplateByDb(num).execute((RedisCallback<Set<String>>) connection -> {

            while (cursor.hasNext()) {
                keysTmp.add(new String(cursor.next()));
            }
            return keysTmp;
        });
        Iterator<String> it = keys.iterator();
        while(it.hasNext()){

            System.out.println(keyStr);

        }
    }

     * 普通缓存获取

     * @return 值
     */
    public Object get(String key,Integer num) {
        return key == null ? null : redisConfig.getRedisTemplateByDb(num).opsForValue().get(ProjectConstant.REDISKEY+key);
    }

    /**

     * @return

    public String getStr(String key,Integer num){

    }

    /**
     * 普通缓存放入

     * @param value 值
     * @return true成功 false失败
     */
    public boolean set(String key, Object value,Integer num) {
        try {
            redisConfig.getRedisTemplateByDb(num).opsForValue().set(ProjectConstant.REDISKEY+key, value);
            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }

    /**
     * 普通缓存放入并设置时间
     * @param key 键
     * @param value 值

        try {

/**
 * Redis配置类
 *
 * @author julian
 * @since 2024/3/20 15:53
 */
@Slf4j
@Component
public class RedisConfig {

    @Value("${spring.redis.cluster.nodes:}")
    private List<String> redisServerNodes;

    @Value("${spring.redis.password}")
    private String redisPassword;

    @Value("${spring.redis.host}")
    private String redisHost;

    @Value("${spring.redis.port}")
    private Integer redisPort;

    public static final int REDIS_DEFAULT_DB = 0;

    public static final int  CLUSTER_MAX_REDIRECTS = 12;

    private static RedisTemplate<Serializable, Object> redisTemplate;


    /**
     * 注入redis集群连接工厂
     * @return 集群连接工厂类JedisConnectionFactory
     */
    @Bean
    @ConditionalOnProperty(prefix = "spring.redis.cluster", name = "nodes")
    public RedisConnectionFactory redisClusterConnectionFactory() {
        log.info("---------redis缓存集群模式已生效----------");
        RedisClusterConfiguration config = new RedisClusterConfiguration(redisServerNodes);
        config.setPassword(RedisPassword.of(redisPassword));
        config.setMaxRedirects(CLUSTER_MAX_REDIRECTS);
        JedisConnectionFactory jedisConnectionFactory = new JedisConnectionFactory(config);
        return jedisConnectionFactory;
    }

    /**
     * 注入redis单机连接工厂
     * @return 单机连接工厂类JedisConnectionFactory
     */
    @Bean
    @ConditionalOnMissingBean(RedisConnectionFactory.class)
    public RedisConnectionFactory redisStandaloneConnectionFactory() {
        log.info("---------redis缓存单机模式已生效----------");
        // 单机版jedis
        RedisStandaloneConfiguration redisStandaloneConfiguration = new RedisStandaloneConfiguration();
        // 设置redis服务器的host或者ip地址
        redisStandaloneConfiguration.setHostName(redisHost);
        // 设置默认使用的数据库
        redisStandaloneConfiguration.setDatabase(REDIS_DEFAULT_DB);
        // 设置密码
        redisStandaloneConfiguration.setPassword(RedisPassword.of(redisPassword));
        // 设置redis的服务的端口号
        redisStandaloneConfiguration.setPort(redisPort);
        JedisConnectionFactory jedisConnectionFactory = new JedisConnectionFactory(redisStandaloneConfiguration);
        return jedisConnectionFactory;
    }

    /**
     * 注入redis模版
     * springboot2.0后默认使用Lettuce，但lettuce6.3以下版本存在Redis异常宕机导致没有回复RST，详见https://github.com/lettuce-io/lettuce-core/issues/2082
     * 基于当前的springboot2.2.6版本，推荐使用jedis进行连接
     * @param redisConnectionFactory redis连接工厂，可以是集群连接工厂，或单节点连接工厂
     * @return redis模版
     */
    @Bean
    public RedisTemplate<Serializable, Object> redisTemplate(RedisConnectionFactory redisConnectionFactory) {
        RedisTemplate<Serializable, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(redisConnectionFactory);
        setSerializer(template);
        this.redisTemplate = template;
        return template;
    }


    /**
     * 设置序列化策略
     * @param template 缓存执行模版
     */
    private void setSerializer(RedisTemplate<Serializable, Object> template) {
        template.setKeySerializer(new StringRedisSerializer());
        template.setValueSerializer(getRedisSerializer());
        template.setHashValueSerializer(getRedisSerializer());
        template.setHashKeySerializer(getRedisSerializer());
    }

    /**
     * 获取通用的换成序列化策略
     * @return 序列化策略
     */
    private RedisSerializer<Object> getRedisSerializer() {
        ObjectMapper objectMapper = new ObjectMapper();
        objectMapper.disable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS);
        objectMapper.registerModule(new JavaTimeModule());
        objectMapper.registerModule(new SimpleModule());
        objectMapper.enableDefaultTyping(ObjectMapper.DefaultTyping.NON_FINAL, JsonTypeInfo.As.PROPERTY);
        GenericJackson2JsonRedisSerializer serializer = new GenericJackson2JsonRedisSerializer(objectMapper);
        return serializer;
    }

    public RedisTemplate<Serializable, Object> getRedisTemplateByDb(int db){
        if(db != 0){
            throw new IllegalArgumentException("为保证redis集群和单机两种模式同时兼容，禁止使用0以外的缓存库，请合并所有db至db0中并检查修复可能存在的key覆盖问题！");
        }
        return redisTemplate;
    }

    public RedisTemplate<Serializable, Object> getRedisTemplate(){
        return redisTemplate;
    }
}
